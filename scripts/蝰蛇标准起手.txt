using System;
using System.Collections.Generic;
using PromeRotation.Core;
using PromeRotation.Data;
using PromeRotation.Extensions;
using PromeRotation.Helpers;
using PromeRotation.Managers;
using PromeRotation.Rotation;
using PRCore = PromeRotation.Core.Core;

// 独立导入版：ViperACR 当前加载时，蛇行交由 ACR 默认功能处理，避免重复。
// 100级标准起手，12个GCD；GCD等待宿主校验。
public sealed class VPR_100_Standard : IOpener
{
    public string OpenerName => "蝰蛇标准起手（500ms蛇行）";
    public bool UseLongArms { get; set; } = true; // 仅其他ACR下的独立起手；Displaced使用UI开关
    public bool UnlimitedDashDistance { get; set; } = false;

    public List<PAction> InCombatSequence
    {
        get
        {
            // ViperACR owns the phased opener so both coils can use live positional data.
            var rotation = RotationManager.GetCurrentRotation();
            if (rotation?.GetType().FullName == "ViperACR.Host.ViperRotation")
                return rotation.GetOpener()?.InCombatSequence ?? new List<PAction>();
            var result = new List<PAction>();
            void Gcd(uint id) => result.Add(new PAction(id, ActionType.Gcd, id == 34626 ? ActionTargetType.Self : ActionTargetType.Target) { RequiresVerification = true });
            void Ability(uint id) => result.Add(new PAction(id, ActionType.OffGcd, ActionTargetType.Target));
            Gcd(34607); // 1G：穿裂尖齿
            result.Add(new PAction(34647, ActionType.OffGcd, ActionTargetType.Self)); // 蛇灵气
            Gcd(34609); // 2G：疾速利齿
            Gcd(34620); // 3G：强碎灵蛇
            Gcd(34621); Ability(34636); Ability(34637); // 4G：猛袭盘蛇、双牙连击、双牙乱击
            Gcd(34622); Ability(34637); Ability(34636); // 5G：疾速盘蛇、双牙乱击、双牙连击
            Gcd(34626); // 6G：祖灵降临
            Gcd(34627); Ability(34640); // 7G
            Gcd(34628); Ability(34641); // 8G
            Gcd(34629); Ability(34642); // 9G
            Gcd(34630); Ability(34643); // 10G
            Gcd(34631); // 11G：祖灵大蛇牙
            Gcd(34613); Ability(34634); // 12G：背裂獠齿、蛇尾击
            foreach (var action in result) action.ExecutionGuard = () => WithinAttackRange(action.ActionId);
            return result;
        }
    }

    private bool WithinAttackRange(uint actionId)
    {
        var me = PRCore.Me;
        var target = PRCore.Target;
        if (me == null || me.IsDead || target == null || target.IsDead || !target.IsTargetable) return false;
        var distance = Math.Max(0, System.Numerics.Vector2.Distance(
            new(me.Position.X, me.Position.Z), new(target.Position.X, target.Position.Z)) - target.HitboxRadius);
        var nativeRange = actionId is 34647 || actionId < 10000 ? 0
            : actionId == 34646 ? 20
            : actionId is 34632 or 34633 or 34644 or 34645 ? 20
            : actionId is 34626 or >= 34634 and <= 34643 ? 5 : 3;
        return float.IsFinite(distance) && (actionId == 34646 && UnlimitedDashDistance ||
            nativeRange == 0 || distance <= nativeRange + (UseLongArms ? 3 : 0));
    }

    public void InitializeCountdown(CountDownHandler countdownHandler)
    {
        // 倒计时剩余500ms：蛇行。使用当前目标；技能本身无伤害。
        countdownHandler.AddAction(500, () =>
        {
            var me = PRCore.Me;
            var target = PRCore.Target;
            var remaining = GameData.GetCountdown();
            return RotationManager.GetCurrentRotation()?.GetType().FullName != "ViperACR.Host.ViperRotation" &&
                PromeSettings.Instance.EnableAcr == AcrState.On && remaining > 0 && remaining <= 0.5f &&
                me != null && !me.IsDead && me.ClassJob.RowId == 41 && me.Level >= 40 &&
                !GameData.IsInCombat() && !GameData.IsInPvPArea() && !GameData.IsPlayerOccupied() &&
                target != null && !target.IsDead && target.IsTargetable && target.CanUseAttackActionOn() &&
                !TargetHelper.TargetHasInvincibilityStatus(target) &&
                WithinAttackRange(34646) &&
                34646u.IsActionAvailableByLevelAndQuest() && 34646u.GetActionCharges() > 0 &&
                ActionHelper.IsReady(34646) && ActionHelper.CanCast(34646, target)
                ? new PAction(34646, ActionType.OffGcd, ActionTargetType.Target) { NetworkTid = target.EntityId }
                : null!;
        });
    }
}
